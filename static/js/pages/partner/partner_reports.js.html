// === PARTNER REPORTS === // ============================================================ // ИСХОДНЫЕ ДАННЫЕ // Передаются шаблоном reports.html через meta-теги // (статические JS не рендерятся шаблонизатором) // ============================================================ function reportsMetaContent(name) { const meta = document.querySelector('meta[name="' + name + '"]'); return meta ? meta.content : ''; } function getReportsCsrfToken() { const input = document.querySelector('[name=csrfmiddlewaretoken]'); if (input) return input.value; const match = document.cookie.match(/(^|;\s*)csrftoken=([^;]+)/); return match ? decodeURIComponent(match[2]) : ''; } const serverSalesData = JSON.parse(reportsMetaContent('sales-graph-data') || '{}'); const trafficSourcesData = JSON.parse(reportsMetaContent('traffic-sources-data') || '{}'); // ============================================================ // УТИЛИТЫ ДАТ // ============================================================ function pad(n) { return String(n).padStart(2, '0'); } function formatDateLabel(date) { // "18.07" return `${pad(date.getDate())}.${pad(date.getMonth() + 1)}`; } function formatDateISO(date) { // "2026-07-18" return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`; } function setDisplayFromHidden(hiddenId, displayId) { var hidden = document.getElementById(hiddenId); var display = document.getElementById(displayId); if (!hidden || !display) return; if (hidden.value) { var parts = hidden.value.split('-'); if (parts.length === 3) { display.value = `${parts[2]}.${parts[1]}.${parts[0]}`; } else { display.value = hidden.value; } } else { display.value = ''; } } function initReportCalendars() { if (typeof CustomCalendar === 'undefined') { console.error('CustomCalendar не найден. Календарь не инициализирован.'); return; } let startCalendar = null; let endCalendar = null; startCalendar = new CustomCalendar('filterStartDisplay', 'filterStartCalendar', 'filterStart', { displayDateFormat: 'dd.mm.yyyy', valueDateFormat: 'yyyy-mm-dd', allowPastDates: true, onSelectDate: function(date) { const startValue = date.getTime(); setDisplayFromHidden('filterStart', 'filterStartDisplay'); } } } }); // Календари для формы генерации отчёта new CustomCalendar('reportStartDisplay', 'reportStartCalendar', 'reportStart', { displayDateFormat: 'dd.mm.yyyy', valueDateFormat: 'yyyy-mm-dd', allowPastDates: true }); new CustomCalendar('reportEndDisplay', 'reportEndCalendar', 'reportEnd', { displayDateFormat: 'dd.mm.yyyy', valueDateFormat: 'yyyy-mm-dd', allowPastDates: true }); } /** * Возвращает массив дат [start..end] включительно. */ function buildDateRange(start, end) { const labels = []; const keys = []; const cur = new Date(start); cur.setHours(0, 0, 0, 0); const stop = new Date(end); stop.setHours(0, 0, 0, 0); while (cur <= stop) { labels.push(formatDateLabel(cur)); keys.push(formatDateISO(cur)); cur.setDate(cur.getDate() + 1); } return { labels, keys }; } /** * Данные для графика: даты + суммы (0, если нет продаж в этот день). */ function buildChartData(start, end) { const { labels, keys } = buildDateRange(start, end); const amounts = keys.map(k => Number(serverSalesData[k] || 0)); return { labels, amounts }; } /** * Красивая шкала Y: целые деления, «ровный» максимум. * Если продаж нет — 0..1000 с шагом 100. */ function calcYAxis(amounts) { const max = Math.max(...amounts, 0); if (max === 0) { return { max: 1000, stepSize: 100 }; } const magnitude = Math.pow(10, Math.floor(Math.log10(max))); const niceMax = Math.ceil(max / magnitude) * magnitude; const stepSize = Math.max(1, Math.round(niceMax / 8)); return { max: niceMax, stepSize }; } // ============================================================ // ГРАФИК ПРОДАЖ // ============================================================ const salesCtx = document.getElementById('salesChart').getContext('2d'); ctx.fillText('о трафике', cx, cy + 10); ctx.restore(); } }; // Настройки для «пустого» состояния (сплошной оранжевый круг) const isTrafficEmpty = totalTraffic === 0; const trafficConfig = isTrafficEmpty ? { labels: ['Нет данных'], data: [1], colors: ['rgba(255, 131, 72, 1)'] } : { labels: sources, data: sourceAmounts, colors: [ 'rgba(255, 140, 66, 0.85)', 'rgba(54, 162, 235, 0.75)', 'rgba(255, 206, 86, 0.75)', 'rgba(75, 192, 192, 0.75)', 'rgba(153, 102, 255, 0.75)', 'rgba(255, 99, 132, 0.75)' ] }; const trafficSourcesChart = new Chart(trafficCtx, { type: 'doughnut', data: { labels: trafficConfig.labels, datasets: [{ data: trafficConfig.data, backgroundColor: trafficConfig.colors, borderColor: '#fff', borderWidth: 2 }] }, options: { responsive: true, maintainAspectRatio: false, cutout: '65%', plugins: { legend: { display: !isTrafficEmpty, position: 'right', labels: { font: { size: 12 }, color: '#333', boxWidth: 14 } }, tooltip: { enabled: !isTrafficEmpty, callbacks: { label: function(context) { const label = context.label || ''; const value = context.raw || 0; const total = context.dataset.data.reduce((a, b) => a + b, 0); const percentage = total ?
